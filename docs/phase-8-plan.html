📄 Example: CaseDataHandler state machine test 
public class CaseStatusMachineTests
{
    private readonly Mock<IApplicationDbContext> _ctx;
    private readonly CaseDataHandler _sut;

    public CaseStatusMachineTests()
    {
        _ctx = new Mock<IApplicationDbContext>();
        _sut = new CaseDataHandler(_ctx.Object);
    }

    [Theory]
    [InlineData("Draft",     "Open",          true)]
    [InlineData("Open",      "InProgress",    true)]
    [InlineData("InProgress", "PendingReview", true)]
    [InlineData("Draft",     "Closed",        false)]  // invalid
    [InlineData("Closed",    "Open",          false)]  // invalid
    public async Task ValidateTransition(
        string from, string to, bool isValid)
    {
        var @case = CaseFixture.Build(status: from);
        _ctx.SetupDbSet(@case);

        if (isValid)
        {
            var result = await _sut.TransitionStatusAsync(@case.CaseNumber, to);
            result.Status.Should().Be(to);
        }
        else
        {
            var act = () => _sut.TransitionStatusAsync(@case.CaseNumber, to);
            await act.Should().ThrowAsync<ValidationException>();
        }
    }
}

📄 Example: ApiFactory with Testcontainers 
public class ApiFactory : WebApplicationFactory<Program>, IAsyncLifetime
{
    private readonly MsSqlContainer _db = new MsSqlBuilder().Build();

    public async Task InitializeAsync()
    {
        await _db.StartAsync();
    }

    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureServices(services =>
        {
            // Replace real DB with Testcontainer connection string
            services.RemoveAll<DbContextOptions<ApplicationDbContext>>();
            services.AddDbContext<ApplicationDbContext>(opt =>
                opt.UseSqlServer(_db.GetConnectionString()));

            // Replace blob with mock
            services.RemoveAll<IBlobStorageService>();
            services.AddSingleton(Mock.Of<IBlobStorageService>(b =>
                b.GetSasUrl(It.IsAny<string>()) ==
                "https://blob.test/file?sv=2024&se=+1h"));
        });
    }

    public new async Task DisposeAsync() => await _db.DisposeAsync();
}

Security Headers

💡
Analogy: Security headers are like putting a "No photography" sign on a sensitive building — they don't stop a determined attacker but eliminate an entire class of opportunistic browser-based attacks (clickjacking, MIME sniffing, XSS via inline scripts) with zero application logic.

🔧 Add Middleware to Program.cs

// In Program.cs, after app.UseHttpsRedirection():
app.Use(async (context, next) =>
{
    context.Response.Headers.Append(
        "Content-Security-Policy",
        "default-src 'self'; script-src 'self'; " +
        "style-src 'self' 'unsafe-inline'; img-src 'self' data:; " +
        "connect-src 'self' https://*.applicationinsights.azure.com; " +
        "frame-ancestors 'none';");
    context.Response.Headers.Append("X-Frame-Options", "DENY");
    context.Response.Headers.Append("X-Content-Type-Options", "nosniff");
    context.Response.Headers.Append("Referrer-Policy", "no-referrer");
    context.Response.Headers.Append(
        "Permissions-Policy",
        "camera=(), microphone=(), geolocation=()");
    await next();
});
HeaderValueWhat it prevents
Content-Security-Policydefault-src 'self'XSS via injected inline scripts / external resources
X-Frame-OptionsDENYClickjacking (embedding in iframe)
X-Content-Type-OptionsnosniffMIME-type sniffing attacks
Referrer-Policyno-referrerLeaking URL path (may contain IDs) to third parties
Permissions-Policycamera=(), microphone=()Prevents rogue JS from requesting device access
Add security headers middleware to Program.cs
CSP + X-Frame-Options + X-Content-Type-Options + Referrer-Policy + Permissions-Policy
Low
